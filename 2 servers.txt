BACKEND SERVER Laravel (PHP)

# Navigate to the Laravel backend
cd c:\xampp\htdocs\appbac\laravel-backend

# First-time setup (if vendor/ is missing)
composer install
copy .env.example .env
php artisan key:generate

# Create MySQL database "appbac" in phpMyAdmin first, then:
php artisan migrate --seed

# Start the API server (port 8000)
php artisan serve --host=0.0.0.0 --port=8000

FRONTEND SERVER Vite/React

# Navigate to the frontend directory
cd c:\xampp\htdocs\appbac\frontend

# Install Node.js dependencies (if not already installed)
npm install

# Start the Vite development server (proxies /api to Laravel on :8000)
npm run dev
